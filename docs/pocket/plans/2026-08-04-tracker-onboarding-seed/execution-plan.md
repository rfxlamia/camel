   ```sql

   -- Onboarding seed: three example items per existing workspace, once (2026-08-04).
   ALTER TABLE workspaces ADD COLUMN IF NOT EXISTS tracker_onboarding_seeded_at TIMESTAMPTZ;

   DO $$
   DECLARE
     ws RECORD;
     v_status_id INTEGER;
     v_key_number INTEGER;
     v_item_id INTEGER;
     v_prefix TEXT;
     v_parts TEXT[];
   BEGIN
     FOR ws IN SELECT id, name FROM workspaces WHERE tracker_onboarding_seeded_at IS NULL LOOP
       SELECT id INTO v_status_id
       FROM tracker_vocabularies
       WHERE workspace_id = ws.id AND kind = 'status'
       ORDER BY (lower(name) = 'backlog') DESC, position ASC, id ASC
       LIMIT 1;

       IF v_status_id IS NULL THEN
         CONTINUE;
       END IF;

       -- Mirrors derivePrefix() (server/src/core/tracker-key.ts): one word -> first
       -- two chars; two+ words -> first letter of the first two words.
       v_parts := array_remove(regexp_split_to_array(trim(ws.name), '\s+'), '');
       IF array_length(v_parts, 1) IS NULL THEN
         v_prefix := '?';
       ELSIF array_length(v_parts, 1) = 1 THEN
         v_prefix := upper(left(v_parts[1], 2));
       ELSE
         v_prefix := upper(left(v_parts[1], 1) || left(v_parts[2], 1));
       END IF;

       UPDATE workspaces SET tracker_key_counter = tracker_key_counter + 1
         WHERE id = ws.id RETURNING tracker_key_counter INTO v_key_number;
       INSERT INTO tracker_items (workspace_id, key_number, title, description, status_id)
       VALUES (
         ws.id, v_key_number, 'Welcome to Tracker',
         'A place for product and backlog work — the kind that spans weeks, not a single day.',
         v_status_id
       )
       RETURNING id INTO v_item_id;
       INSERT INTO tracker_events (tracker_item_id, actor_id, event_type, payload, workspace_id)
       VALUES (
         v_item_id, NULL, 'tracker_item_created',
         jsonb_build_object('title', 'Welcome to Tracker', 'key', v_prefix || '-' || v_key_number::text),
         ws.id
       );

       UPDATE workspaces SET tracker_key_counter = tracker_key_counter + 1
         WHERE id = ws.id RETURNING tracker_key_counter INTO v_key_number;
       INSERT INTO tracker_items (workspace_id, key_number, title, description, status_id)
       VALUES (
         ws.id, v_key_number, 'This isn''t your Board',
         'One item here (like renaming a workspace) can spawn several Board cards over weeks. Different granularity, not just a different screen.',
         v_status_id
       )
       RETURNING id INTO v_item_id;
       INSERT INTO tracker_events (tracker_item_id, actor_id, event_type, payload, workspace_id)
       VALUES (
         v_item_id, NULL, 'tracker_item_created',
         jsonb_build_object('title', 'This isn''t your Board', 'key', v_prefix || '-' || v_key_number::text),
         ws.id
       );

       UPDATE workspaces SET tracker_key_counter = tracker_key_counter + 1
         WHERE id = ws.id RETURNING tracker_key_counter INTO v_key_number;
       INSERT INTO tracker_items (workspace_id, key_number, title, description, status_id)
       VALUES (
         ws.id, v_key_number, 'Try it yourself',
         'Compare this list with your Board, then hit "New item" to create your first one.',
         v_status_id
       )
       RETURNING id INTO v_item_id;
       INSERT INTO tracker_events (tracker_item_id, actor_id, event_type, payload, workspace_id)
       VALUES (
         v_item_id, NULL, 'tracker_item_created',
         jsonb_build_object('title', 'Try it yourself', 'key', v_prefix || '-' || v_key_number::text),
         ws.id
       );

       UPDATE workspaces SET tracker_onboarding_seeded_at = now() WHERE id = ws.id;
     END LOOP;
   END $$;
   ```

   `tracker_events.payload` includes both `title` and `key` (formatted like the live route's `CAM-1`), matching the spec's Rule 1/Rule 4 examples exactly — `v_prefix` is computed once per workspace by replicating `derivePrefix()`'s word-splitting logic in PL/pgSQL, since the key string is derived at read time everywhere else in this codebase and no stored prefix column exists to read from instead.
